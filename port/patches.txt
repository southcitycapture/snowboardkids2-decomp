# The ollie's launch vector, which the PowerPC linker took apart.
#
# beginPostTrickLaunchStep (race_main.c ~1911) builds the jump impulse in three
# separate file-scope globals and then hands the *address of the first one* to
# transformVector2, which reads three consecutive s32 as a Vec3i:
#
#     D_800BAB40_AA9F0 = player->unkB8C + player->baseAcceleration;   /* y */
#     D_800BAB44_AA9F4 = 0x20000;                                     /* z */
#     transformVector2(&D_800BAB3C_AA9EC, &rotationTemp2, &launchVelocity);
#
# On the N64 the three sit at 0xBAB3C, 0xBAB40, 0xBAB44 in declaration order and
# that pun is exactly a Vec3i. In the Mach-O build they are three independent
# zero-initialised globals, and `nm` puts them at 0xb428, 0xb424 and 0xb420 --
# *descending*. So the transform read the x it wanted and then two words past
# the end of the block, and every jump in the port, by every rider on every
# course, was launched with whatever those two words happened to hold.
#
# It is not a subtle wrongness: on X Cross the CPU rider's ollie came out at a
# fixed +24,000 of vertical velocity against a slope carrying it down at
# -29,000, so it never left the ground, updatePostTrickDescentStep's
# behaviourCounter ran out every time, and ninety-two jumps in a ninety-second
# race scored nothing. Multiplying baseAcceleration by five changed the launch
# by not one unit -- which is the measurement that says the field is not being
# read at all.
#
# The three become one array and keep their names, so the pun is a real Vec3i
# and nothing else in the file has to change.
src/race/race_main.c	s32 D_800BAB3C_AA9EC = 0;	s32 sbk_ollie_vec[3];\n#define D_800BAB3C_AA9EC sbk_ollie_vec[0]
src/race/race_main.c	s32 D_800BAB40_AA9F0 = 0;	#define D_800BAB40_AA9F0 sbk_ollie_vec[1]
src/race/race_main.c	s32 D_800BAB44_AA9F4 = 0;	#define D_800BAB44_AA9F4 sbk_ollie_vec[2]

# The rest of the ollie bug: the audit. See port/docs/PLAN.md, "The audit for
# the ollie bug's siblings", for the method and the full result.
#
# The short version: the pun only breaks where the globals are UNPINNED. A
# pinned global keeps its N64 address in the emulated RDRAM (port/tools/
# gen_pins.py) and the code reads it through that absolute symbol, so its
# neighbours are still its N64 neighbours and any amount of reading across the
# boundary works exactly as the original did. It is the 248 unpinned globals
# -- the overlay range and what the pin list does not cover -- that the Mach-O
# linker is free to reorder, and it did: descending, in run after run.
#
# Two of them are read across, and this is the second one.

# gCharacterEffectSpawnPointsBoard is the four corners of a snowboard, and the
# decomp has it as s16[16] -- half a corner short. updateSlotAnimation loops
# i = 0..3 calling transformVector(&gCharacterEffectSpawnPointsBoard[i * 6]),
# and transformVector reads a whole Vec3i, so the four calls read 24 s16 out
# of an array of 16. On the N64 the eight that follow are
# gShortcutChanceByMemoryPool's first sixteen bytes, and read as s32 they are
# exactly (-6, 0, 28) and (-6, 0, -28): the two back corners. In the port
# gShortcutChanceByMemoryPool is 0x10 *below* the array and
# gCharacterEffectSpawnPoints is immediately above it, so two of the four
# spark points come out at the wrong place. The shortcut table itself is left
# where it is -- nothing writes either, so the eight overlapping s16 are
# simply written out twice.
src/animation/slot_animation.c	s16 gCharacterEffectSpawnPointsBoard[16] = {	s16 sbk_effect_spawn_block[24] = {\n    6, 0, 0, 0, 28, 0, 6, 0, 0, 0, -28, 0, -6, 0, 0, 0,\n    /* the two corners the N64 read out of gShortcutChanceByMemoryPool: */\n    28, 0, -6, 0, 0, 0, -28, 0,\n};\n#define gCharacterEffectSpawnPointsBoard sbk_effect_spawn_block\ns16 sbk_effect_spawn_board_orig[] = {

# The speed cap the game actually uses, so the boss race's governor can reach it.
#
# race_main.c:802 rewrites `maxSpeedCap` from `baseMaxSpeed` every frame, and
# updateIceLandBoss rewrites the boss's from its own distance branch every
# frame, both *before* the clamp that consumes it -- so a host tick writing
# either field between frames changes nothing at all. Three separate
# measurements came back byte-identical to the run before them (the pace
# governor, the knockback breaker, --bossslow) before the reason was read off
# `clampPlayerVelocityToMaxSpeed` (track_height.c:61), which is the one place
# the number is used: `maxMagnitude = player->maxSpeedCap`.
#
# So the hook goes there, one line, immediately before the read. It is the
# port's only way to govern a rider's speed at the moment it takes effect, and
# it is inert unless --autoplay has armed the boss-race governor for exactly
# the two riders in a health-boss race (port/src/debug/boss_pilot.c).
src/race/track_height.c	maxMagnitude = player->maxSpeedCap;	{ extern s32 sbk_speed_cap(Player *, s32); player->maxSpeedCap = sbk_speed_cap(player, player->maxSpeedCap); }\n    maxMagnitude = player->maxSpeedCap;
